import sqlite3

conn = sqlite3.connect("test.db")
cursor = conn.cursor()

cursor.execute("""
    CREATE TABLE IF NOT EXISTS movies (
    id INTEGER PRIMARY KEY,
    name TEXT, 
    type TEXT,
    year INTEGER
);
""")

cursor.execute('''
    CREATE TABLE IF NOT EXISTS ice_cream (
        id INTEGER PRIMARY KEY,
        name TEXT,
        type TEXT,
        title TEXT,
        description TEXT,
        price REAL,
        is_published INTEGER,
        is_on_main INTEGER
    );
''')


ice_cream = [
    (1, 'Егор', 'Мужчина', 'Карамельное', 'Очень вкусное 5 из 10', 250, 1, 1),
    (2, 'Вася', 'Девушка', 'Шоколадное', 'Вкусное 4 из 10', 400, 1, 1),

    (4, 'Иван', 'Мужчина', 'Клубничное', 'Вкусное 4 из 10', 150, 1, 0),
]
movies = [
    (1, "Весёлые мелодии", "Мультсериал", 1930, None),

    (6, "Она написала убийство", "сериал", 1984, None),
    (7, "Лас-Вегас", "сериал", 2003, None),
]

cursor.execute('''
    ALTER TABLE movies ADD COLUMN gross INTEGER DEFAULT NULL;   ''')

cursor.executemany(""" INSERT OR IGNORE INTO movies VALUES (?, ?, ?, ?, ?)""", movies)
