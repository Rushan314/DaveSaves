import sqlite3

# Подключаемся к базе данных
conn = sqlite3.connect("name.db")
cursor = conn.cursor()

# 1. Безопасное удаление старых таблиц перед пересозданием
cursor.execute("DROP TABLE IF EXISTS ice_cream_sorted;")
cursor.execute("DROP TABLE IF EXISTS wrappers;")
cursor.execute("DROP TABLE IF EXISTS categories;")
cursor.execute("DROP TABLE IF EXISTS ice_cream;")

# 2. Создание основных таблиц
cursor.execute('''
CREATE TABLE ice_cream (
    id INTEGER PRIMARY KEY,
    name TEXT NOT NULL,
    price INTEGER
);
''')

cursor.execute('''
CREATE TABLE wrappers (
    id INTEGER PRIMARY KEY,
    name TEXT NOT NULL,
    original_name_id INTEGER NOT NULL UNIQUE,
    FOREIGN KEY(original_name_id) REFERENCES ice_cream(id)
);
''')

cursor.execute('''
CREATE TABLE categories (
    id INTEGER PRIMARY KEY,
    slug TEXT NOT NULL,
    categories_id INTEGER NOT NULL UNIQUE,
    FOREIGN KEY(categories_id) REFERENCES ice_cream(id)
);
''')

# 3. Подготовка и вставка исходных данных
ice_cream = [
    (1, 'Ice Action Hero', 123),
    (2, 'ice, She Wrote', 121),
    (3, 'ice Tunes', 100),
    (4, 'Il ice, il brutto, il cattivo', 1233),
    (5, 'ice Framed Roger Rabbit', 2993),
    (6, 'Merrie ice', 1203)
]

wrappers = [
    (1, 'Обертка1', 3),
    (2, 'Обертка2', 6),
    (3, 'Обертка3', 5),
    (4, 'Бумажная с черепами', 4),
    (5, 'Обертка5', 1),
    (6, 'Обертка6', 2)
]

categories = [
    (1, '«Экзотические»', 3),
    (2, '«Экзотические»', 6),
    (3, '«Экзотические»', 5),
    (4, '«Классические»', 4),
    (5, '«Экзотические»', 1),
    (6, '«Классические»', 2)
]

cursor.executemany('INSERT INTO ice_cream VALUES(?, ?, ?);', ice_cream)
cursor.executemany('INSERT INTO wrappers VALUES(?, ?, ?);', wrappers)
cursor.executemany('INSERT INTO categories VALUES(?, ?, ?);', categories)

# 4. Добавление новых столбцов в таблицу ice_cream
cursor.execute('''
ALTER TABLE ice_cream
ADD COLUMN is_published INTEGER NOT NULL DEFAULT 1;
''')

cursor.execute('''
ALTER TABLE ice_cream
ADD COLUMN is_on_main INTEGER NOT NULL DEFAULT 0;
''')

# 5. Создание новой таблицы из отсортированных данных (по убыванию цены)
cursor.execute('''
CREATE TABLE ice_cream_sorted AS
SELECT * FROM ice_cream
ORDER BY price DESC;
''')

# Сохраняем все изменения в базе
conn.commit()

# 6. Проверка: выводим содержимое новой отсортированной таблицы
cursor.execute("SELECT * FROM ice_cream_sorted;")
results = cursor.fetchall()

print("Содержимое новой таблицы ice_cream_sorted:")
for row in results:
    print(row)

# 7. Закрываем курсор и соединение в самом конце
cursor.close()
conn.close()
